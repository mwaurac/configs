## Layout

```text
init.lua
lua/config/options.lua   # editor options
lua/config/keymaps.lua   # general keymaps
lua/config/autocmds.lua  # autocmds
lua/config/lazy.lua      # lazy.nvim bootstrap
lua/plugins/theme.lua       # gruvbox dark (default)
lua/plugins/explorer.lua    # neo-tree side panel
lua/plugins/harpoon.lua     # harpoon v2
lua/plugins/tabs.lua        # bufferline + lualine + bufremove
lua/plugins/lsp.lua         # lua / python / rust / c-c++ (mason)
lua/plugins/session.lua     # per-directory sessions (persistence)
lua/plugins/trouble.lua     # diagnostics / symbols lists
lua/plugins/rust.lua        # Cargo.toml versions/upgrades (crates.nvim)
lua/plugins/markview.lua    # markdown preview (<leader>p)
lua/plugins/completion.lua  # blink.cmp (+ `:` autosuggest)
lua/plugins/noice.lua       # noice popup cmdline + notify
lua/plugins/treesitter.lua  # treesitter `main` branch (needs `tree-sitter` CLI: sudo pacman -S tree-sitter-cli)
lua/plugins/telescope.lua
lua/plugins/editor.lua      # which-key, gitsigns, autopairs, comment, conform
lua/plugins/ui.lua          # dashboard, indent guides
```
